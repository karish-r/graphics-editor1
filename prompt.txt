Response:
Sure! Here's a simple standalone program that demonstrates only the drawRectangle() function.

Rectangle Drawing Program
#include <stdio.h>

#define ROWS 10
#define COLS 30

char canvas[ROWS][COLS];

/* Fill canvas with '_' */
void initializeCanvas()
{
    for(int i = 0; i < ROWS; i++)
    {
        for(int j = 0; j < COLS; j++)
        {
            canvas[i][j] = '_';
        }
    }
}

/* Display canvas */
void displayCanvas()
{
    for(int i = 0; i < ROWS; i++)
    {
        for(int j = 0; j < COLS; j++)
        {
            printf("%c", canvas[i][j]);
        }
        printf("\n");
    }
}

/* Draw filled rectangle */
void drawRectangle(int x, int y, int width, int height)
{
    for(int i = y; i < y + height; i++)
    {
        for(int j = x; j < x + width; j++)
        {
            if(i >= 0 && i < ROWS &&
               j >= 0 && j < COLS)
            {
                canvas[i][j] = '*';
            }
        }
    }
}

int main()
{
    initializeCanvas();

    printf("Canvas Before Drawing:\n\n");
    displayCanvas();

    drawRectangle(3, 2, 8, 4);

    printf("\nCanvas After Drawing Rectangle:\n\n");
    displayCanvas();

    return 0;
}

How It Works

The loops:

for(int i = y; i < y + height; i++)
{
    for(int j = x; j < x + width; j++)

visit every position inside the rectangle and replace:

_

with

*

For your example:

drawRectangle(3,2,8,4);

the rectangle occupies:

Rows:    2 → 5
Columns: 3 → 10

which creates an 8 × 4 filled rectangle on the canvas.
